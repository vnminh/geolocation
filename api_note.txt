[POST] /prediction
func: upload image to server to get prediction for image 
body: formdata
	- image (*)
return:
	- success
	- data:
		- image_url 
		- answer
			- type (duplicate_image | new_image_known_location | new_location_request | mock)
		- request_id (nullable, created when type = new_location_request)
------------------------------------------------------------
[POST] /prediction/stream
func: stream full prediction pipeline (retrieval stage, predict stage, answer tokens)
body: formdata
	- image (*)
return: text/event-stream (SSE)
	events:
	- start: { success, image_url }
	- stage: { name: retrieval|predict, status: started|thinking|completed, ... }
	- final: { success, data: { image_url, answer } }
	- done: { success: true }
	- error: { success: false, detail }
------------------------------------------------------------
